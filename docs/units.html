The Core Collection, the units the package ships, on a language model: sixteen of them, effects that act on the stream, fitted units, a neuron editor and meters. Each is one file in src/vau/units/, and each figure animates the change its code makes on GPT-2 small. The same units, tuned for an audio model, are VAU audio.

Each section below is built from the unit's manifest (its UNIT dict) and its code(u) by tools/build_site.py, which also runs vau.testing.conformance on it. Neutral is the value at which the unit changes nothing; a unit set to all its neutral values leaves the model's output as it was.

def compressor_edit(threshold: float, ratio: float, makeup: float):
    """Above `threshold` (a size), the stream's size at a position is reduced: every unit over
    the threshold becomes 1/ratio of a unit. Then everything is multiplied by `makeup`."""
    def edit(h, pos):
        x = h.float()
        n = x.norm(dim=-1, keepdim=True).clamp_min(1e-9)
        target = torch.where(n > threshold, threshold + (n - threshold) / ratio, n)
        return (x * (target / n) * makeup).to(h.dtype)
    return edit

def saturator_edit(level: torch.Tensor, mix: float):
    """Each dimension soft-clipped at its own `level` (level * tanh(x / level)), mixed with the
    dry stream by `mix`."""
    def edit(h, pos):
        x = h.float()
        wet = level * torch.tanh(x / level)
        return ((1 - mix) * x + mix * wet).to(h.dtype)
    return edit

delay.pyThe function it uses
def delay_edit(time: int, feedback: float, mix: float, line: Dict[int, torch.Tensor], first: int = 1):
    """Adds `mix` times the delay line `time` tokens back. The line at a position is the stream
    there plus `feedback` times the line `time` tokens before it, so echoes repeat and fade.
    `first` is the first position bent: 1 where position 0 is an attention sink, else 0. The line
    keeps only the last `time` positions, all it ever reads again."""
    def edit(h, pos):
        x = h.float()
        out = x.clone()
        for j, p in enumerate(pos.tolist()):
            src = line.get(p - time) if p - time >= first else None
            if p >= first:
                line[p] = (x[0, j] + feedback * src) if src is not None else x[0, j].clone()
            if src is not None:
                out[0, j] = x[0, j] + mix * src
            line.pop(p - time, None)
        return out.to(h.dtype)
    return edit

reverb.pyThe function it uses
def reverb_edit(decay: float, mix: float, tail: Dict[int, torch.Tensor], first: int = 1):
    """Adds `mix` times a decaying average of the earlier positions (the tail): the tail at a
    position is `decay` times the tail before it plus (1 - decay) times the stream there.
    `first` is the first position bent (1 where position 0 is an attention sink, else 0); only
    the last position's tail is kept."""
    def edit(h, pos):
        x = h.float()
        out = x.clone()
        for j, p in enumerate(pos.tolist()):
            prev = tail.get(p - 1) if p - 1 >= first else None
            if p >= first:
                tail[p] = (decay * prev + (1 - decay) * x[0, j]) if prev is not None else x[0, j].clone()
            if prev is not None:
                out[0, j] = x[0, j] + mix * prev
            tail.pop(p - 1, None)
        return out.to(h.dtype)
    return edit

erosion.pyThe function it uses
def erosion_edit(step: Optional[torch.Tensor], hold: int, held: Dict[int, torch.Tensor], first: int = 1):
    """Quantises each dimension to multiples of its own `step` (fewer bits, coarser steps), and
    holds the stream of the first position of each run of `hold` positions over the rest.
    `first` is the first position bent: 1 where position 0 is an attention sink, else 0. Only the
    run in hand is kept."""
    def edit(h, pos):
        x = h.float()
        if step is not None:
            x = torch.round(x / step) * step
        if hold > 1:
            x = x.clone()
            for j, p in enumerate(pos.tolist()):
                if p < first:
                    continue
                start = first + ((p - first) // hold) * hold
                if p == start:
                    held.pop(start - hold, None)
                    held[start] = x[0, j].clone()
                elif start in held:
                    x[0, j] = held[start]
        if (step is not None or hold > 1) and first:
            x = torch.where(pos.view(1, -1, 1) == 0, h.float(), x)   # the attention sink untouched
        return x.to(h.dtype)
    return edit

shifter.pyThe function it uses
def shifter_edit(e1: torch.Tensor, e2: torch.Tensor, angle: float):
    """Rotates the stream by `angle` degrees in the plane of two orthonormal directions e1, e2,
    leaving every other component as it was. 180 degrees negates both components."""
    c, s = math.cos(math.radians(angle)), math.sin(math.radians(angle))
    def edit(h, pos):
        x = h.float()
        a, b = (x * e1).sum(-1, keepdim=True), (x * e2).sum(-1, keepdim=True)
        return (x + (a * c - b * s - a) * e1 + (a * s + b * c - b) * e2).to(h.dtype)
    return edit

eq.pyThe function it uses
def eq_edit(mean: torch.Tensor, bands: List[torch.Tensor], gains: List[float]):
    """Splits the stream (less the corpus mean) into bands of principal directions and scales
    each band by its gain; the mean is kept."""
    def edit(h, pos):
        x = h.float()
        xc = x - mean
        for v, g in zip(bands, gains):
            if g != 1.0:
                x = x + (g - 1.0) * ((xc @ v.T) @ v)
        return x.to(h.dtype)
    return edit

beatrepeat.pyThe function it uses
def beatrepeat_edit(interval: int, length: int, mix: float, kept: Dict[int, torch.Tensor], first: int = 1):
    """Over the last `length` positions of each run of `interval` (from position `first`: 1 where
    position 0 is an attention sink, else 0), the stream is moved toward the stream `length`
    positions back, as it was before this edit. Only the last `length` positions are kept."""
    length = min(length, interval - 1)

    def edit(h, pos):
        x = h.float()
        out = x.clone()
        for j, p in enumerate(pos.tolist()):
            if p < first:
                continue
            kept[p] = x[0, j].clone()
            if (p - first) % interval >= interval - length and (p - length) in kept and p - length >= first:
                out[0, j] = (1 - mix) * x[0, j] + mix * kept[p - length]
            kept.pop(p - length, None)
        return out.to(h.dtype)
    return edit

gate.pyThe function it uses
def gate_edit(keep: float, sink: bool, has_sink: bool = True):
    """Each position's components outside the largest `keep` share (by size) set to zero; the
    first position left alone unless `sink`, or unless the model has no attention sink there
    (`has_sink` False)."""
    def edit(h, pos):
        if keep >= 1.0:
            return h
        x = h.float()
        k = max(1, math.ceil(keep * x.shape[-1]))
        idx = x.abs().topk(k, dim=-1).indices
        out = torch.zeros_like(x).scatter(-1, idx, x.gather(-1, idx))
        if not sink and has_sink:
            out = torch.where(pos.view(1, -1, 1) == 0, x, out)
        return out.to(h.dtype)
    return edit

limiter.pyThe function it uses
def limiter_edit(q: float, sink: bool, has_sink: bool = True):
    """Each position's components clamped to ± the q-quantile of their absolute values; the first
    position (the attention sink) left alone unless `sink`."""
    def edit(h, pos):
        if q >= 1.0:
            return h
        x = h.float()
        lim = torch.quantile(x.abs(), q, dim=-1, keepdim=True)
        out = torch.maximum(torch.minimum(x, lim), -lim)
        if not sink and has_sink:
            out = torch.where(pos.view(1, -1, 1) == 0, x, out)
        return out.to(h.dtype)
    return edit

multiband.pyThe function it uses
def multiband_edit(mean: torch.Tensor, bands: List[torch.Tensor], limits: List[float], ratio: float, which: List[int]):
    """Each chosen band of principal directions compressed above its limit (a size): the band's
    component of the stream (less the mean) is shortened so that every unit over the limit becomes
    1/ratio of a unit."""
    def edit(h, pos):
        x = h.float()
        xc = x - mean
        out = x.clone()
        for b in which:
            v, lim = bands[b], limits[b]
            y = xc @ v.T                                        # the band's coordinates
            n = y.norm(dim=-1, keepdim=True).clamp_min(1e-9)
            target = torch.where(n > lim, lim + (n - lim) / ratio, n)
            out = out + ((y * (target / n - 1)) @ v)
        return out.to(h.dtype)
    return edit

Help

Fitted to the two sets of texts a direction was made from. With move=away what each chosen block writes on the direction's positive texts is pulled toward what it writes on the negative texts (or the reference corpus, for a direction made from a passage or a word); with move=toward the other way. What it writes on the set moved toward, and on the reference corpus, is held where it was (keep); pull weighs the move against that; over pushes the moved outputs from where they were; neighbours is how many of the nearest outputs each is pulled toward; rank is the change's rank. The fit runs once per setting, before the run, and is cached; amount scales the fitted change and needs no new fit. Amount 0 is neutral. The fit works token by token: where a contrast set's paired texts are the same up to the construction ("The door is" before "not"), the positions before it have nothing to pull, so vMorph changes how the model goes on after the construction more than whether it writes it.

morph.pyThe function it uses
def morph_hook(A: torch.Tensor, B: torch.Tensor, scale: Optional[torch.Tensor], bias: Optional[torch.Tensor]):
    """Forward hook on an output projection: adds (x @ A.T) @ B.T to what it writes. With `scale`,
    the module's own product and the addition are scaled per output feature (the bias is not)."""
    def hook(module, args, output):
        y = output[0] if isinstance(output, tuple) else output
        x = args[0].float()
        d = (x @ A.T) @ B.T
        if scale is None:
            new = y.float() + d
        else:
            base = y.float() - bias if bias is not None else y.float()
            new = (base + d) * scale + (bias if bias is not None else 0)
        new = new.to(y.dtype)
        return (new,) + tuple(output[1:]) if isinstance(output, tuple) else new
    return hook


def fit_morph(W, b, kept, moved, toward, p, seed: int) -> Dict[str, torch.Tensor]:
    """A (rank, in) and B (out, rank) for one module. A starts as PyTorch starts a linear layer's
    weight, from a fixed seed; B starts at zero, which is no change at all."""
    out_f, in_f = W.shape
    A = torch.empty(p["rank"], in_f)
    torch.nn.init.kaiming_uniform_(A, a=math.sqrt(5), generator=torch.Generator().manual_seed(seed))
    A.requires_grad_(True)
    B = torch.zeros(out_f, p["rank"], requires_grad=True)
    sub = (lambda Y: Y - b) if b is not None else (lambda Y: Y)
    kept = (kept[0], kept[1], sub(kept[1]))
    moved = (moved[0], moved[1], sub(moved[1]))
    opt = torch.optim.LBFGS([A, B], lr=1.0, max_iter=20, history_size=10, line_search_fn="strong_wolfe")

    def closure():
        opt.zero_grad()
        loss = morph_loss(W, b, kept, moved, toward, A, B, p)
        loss.backward()
        return loss
    with torch.enable_grad():
        for _ in range(STEPS):
            opt.step(closure)
    return {"A": A.detach().clone(), "B": B.detach().clone()}


def morph_loss(W, b, kept, moved, toward, A, B, p) -> torch.Tensor:
    """keep × the mean squared change on the kept set, plus pull × (the moved set's mean distance
    to its nearest original outputs on the set it moves toward, less over × its distance to its
    own original outputs)."""
    Xk, Yk, base_k = kept
    Xm, Ym, base_m = moved
    nk = morph_outputs(W, b, Xk, base_k, A, B, p["size"])
    nm = morph_outputs(W, b, Xm, base_m, A, B, p["size"])
    keep = ((nk - Yk) ** 2).mean()
    pull = knn_distance(nm, toward, p["k"]).mean() - p["over"] * knn_distance(nm, Ym, p["k"]).mean()
    return p["keep"] * keep + p["pull"] * pull


def morph_outputs(W, b, X, base, A, B, keep_size: bool):
    """What the module writes with the change: its own product plus (X @ A.T) @ B.T, each output
    feature scaled back to its weight row's original length if size is kept, then the bias."""
    y = base + (X @ A.T) @ B.T
    if keep_size:
        y = y * (W.norm(dim=1) / (W + B @ A).norm(dim=1).clamp_min(1e-9))
    return y + b if b is not None else y


def knn_distance(a: torch.Tensor, b: torch.Tensor, k: int) -> torch.Tensor:
    """For each row of a, the mean Euclidean distance to its k nearest rows of b."""
    d = torch.cdist(a, b)
    return d.topk(min(k, b.shape[0]), dim=1, largest=False).values.mean(1)

Help

def abliterate_hook(v: torch.Tensor, s: float, scale: Optional[torch.Tensor], bias: Optional[torch.Tensor], with_bias: bool):
    """Forward hook on an output projection: removes s times the output's component along the
    unit direction v. Without the bias, only what the weights write (output - bias) is projected.
    With `scale`, the projected weights' output features are scaled back to their rows' lengths."""
    def hook(module, args, output):
        y = output[0] if isinstance(output, tuple) else output
        yf = y.float()
        b = bias if bias is not None else 0
        w = yf - b                                   # what the weights write
        w = w - s * (w @ v).unsqueeze(-1) * v
        if scale is not None:
            w = w * scale
        if bias is not None:
            b2 = bias - s * (bias @ v) * v if with_bias else bias
        else:
            b2 = 0
        new = (w + b2).to(y.dtype)
        return (new,) + tuple(output[1:]) if isinstance(output, tuple) else new
    return hook


def kernel(n_blocks: int, shape: str, peak: float, width: float, edge: float) -> List[float]:
    """The strength (as a share of the peak's) at each block. Peak and width are shares of the
    depth. Linear: from 1 at the peak straight down to `edge` at `width` away, nothing beyond.
    Bell: a Gaussian whose value at `width` away is `edge` (or a tenth, if edge is 0), cut off
    beyond. Flat: 1 everywhere."""
    last = max(1, n_blocks - 1)
    p, w = peak * last, max(1e-6, width * last)
    out = []
    for l in range(n_blocks):
        d = abs(l - p)
        if shape == "flat":
            out.append(1.0)
        elif d > w + 1e-9:
            out.append(0.0)
        elif shape == "bell":
            e = edge if edge > 0 else 0.1
            out.append(math.exp(math.log(e) * (d / w) ** 2) if e < 1 else 1.0)
        else:
            out.append(1.0 + (d / w) * (edge - 1.0))
    return out


def unit_direction(v: torch.Tensor, mean: Optional[torch.Tensor]) -> torch.Tensor:
    """The unit direction, made orthogonal to the stream's mean first if a mean is given."""
    v = v.float()
    if mean is not None:
        m = mean.float() / mean.float().norm().clamp_min(1e-9)
        v = v - (v @ m) * m
    return v / v.norm().clamp_min(1e-9)

neuron.pyThe function it uses
def neuron_hook(idx: torch.Tensor, op: str, value: float):
    """Forward pre-hook on an MLP's output projection: its input holds each neuron's activation
    at each position; edit the named ones."""
    def hook(module, args):
        x = args[0].clone()
        a = x[..., idx]
        if op == "scale":
            a = a * value
        elif op == "add":
            a = a + value
        elif op == "set":
            a = torch.full_like(a, value)
        elif op == "zero":
            a = torch.zeros_like(a)
        elif op == "flip":
            a = -a
        elif op == "clamp":
            a = a.clamp(-abs(value), abs(value))
        x[..., idx] = a
        return (x,) + tuple(args[1:])
    return hook

def peak_share(stream, k: int):
    """Each row's share of its squared size in its k largest components, in percent."""
    sq = stream.float() ** 2
    return (100 * sq.topk(min(k, sq.shape[-1]), dim=-1).values.sum(-1) / sq.sum(-1).clamp_min(1e-12)).tolist()
